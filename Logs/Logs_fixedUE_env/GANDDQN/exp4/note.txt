def calc_reward(qoe, se, threshold):
    # 依照權重算出 utility
    utility = np.matmul(qoe_weight, qoe.reshape((3, 1))) + se_weight * se 
    
    # # 這演算法的 threshold 是會隨時間而單調上升的，故要限制 threshold
    threshold = 3.5 + 3 * frame / (total_timesteps / 1.25)  # 讓他在 8000 個 episdoes 左右時要求就到達 6.5
    if threshold > 6.5:
        threshold = 6.5

    # reward clipping
    if utility < threshold:
        reward = 0
    else:
        reward = 1
    return utility, reward
